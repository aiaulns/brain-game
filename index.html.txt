  <script>
    // State management
    let state = {
      score: 0,
      soundEnabled: true,
      currentQuizIndex: 0,
      quizTimerInterval: null,
      quizTimeLeft: 20,
      selectedQuizOption: null,
      currentSimIndex: 0,
      selectedMatchItem: null,
      matchAnswers: {}
    };

    // Web Audio API Sound Generator
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playSound(type) {
      if (!state.soundEnabled) return;
      try {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.connect(gain);
        gain.connect(audioCtx.destination);

        if (type === 'correct') {
          osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); // C5
          osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.3); // A5
          gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'wrong') {
          osc.frequency.setValueAtTime(220, audioCtx.currentTime); // A3
          osc.frequency.exponentialRampToValueAtTime(110, audioCtx.currentTime + 0.3); // A2
          gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'click') {
          osc.frequency.setValueAtTime(400, audioCtx.currentTime);
          gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);
          osc.start();
          osc.stop(audioCtx.currentTime + 0.05);
        }
      } catch (e) {
        console.warn('Audio playback error', e);
      }
    }

    function toggleSound() {
      state.soundEnabled = !state.soundEnabled;
      const btn = document.getElementById('sound-btn');
      btn.innerHTML = state.soundEnabled ? 
        '<i class="fa-solid fa-volume-high text-cyan-400"></i>' : 
        '<i class="fa-solid fa-volume-xmark text-slate-500"></i>';
    }

    function toggleFullscreen() {
      if (!document.fullscreenElement) {
        document.documentElement.requestFullscreen();
      } else {
        if (document.exitFullscreen) document.exitFullscreen();
      }
    }

    function updateScore(points) {
      state.score += points;
      document.getElementById('global-score').innerText = state.score;
    }

    function resetGame() {
      state.score = 0;
      document.getElementById('global-score').innerText = '0';
      switchMode('menu');
    }

    function switchMode(mode) {
      playSound('click');
      clearInterval(state.quizTimerInterval);

      document.getElementById('mode-menu').classList.add('hidden');
      document.getElementById('quiz-view').classList.add('hidden');
      document.getElementById('sim-view').classList.add('hidden');
      document.getElementById('match-view').classList.add('hidden');
      document.getElementById('finish-view').classList.add('hidden');

      if (mode === 'menu') document.getElementById('mode-menu').classList.remove('hidden');
      if (mode === 'quiz') document.getElementById('quiz-view').classList.remove('hidden');
      if (mode === 'sim') document.getElementById('sim-view').classList.remove('hidden');
      if (mode === 'match') document.getElementById('match-view').classList.remove('hidden');
      if (mode === 'finish') {
        document.getElementById('finish-view').classList.remove('hidden');
        document.getElementById('final-score').innerText = state.score;
        let rank = "Интерн-Нейробиолог";
        if (state.score >= 500) rank = "Эрудированный Нейрофизиолог";
        if (state.score >= 800) rank = "Главный Нейрохирург & Диагност";
        document.getElementById('final-rank').innerText = `Присвоено звание: ${rank}`;
        confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });
      }
    }

    function loadQuizQuestion() {
      const q = quizData[state.currentQuizIndex];
      document.getElementById('quiz-progress-text').innerText = `Вопрос ${state.currentQuizIndex + 1} из ${quizData.length}`;
      document.getElementById('quiz-progress-bar').style.width = `${((state.currentQuizIndex + 1) / quizData.length) * 100}%`;
      document.getElementById('quiz-question').innerText = q.question;
      
      const optionsContainer = document.getElementById('quiz-options');
      optionsContainer.innerHTML = '';

      q.options.forEach((opt, idx) => {
        const btn = document.createElement('button');
        btn.className = 'w-full text-left p-4 rounded-xl glass-card hover:bg-slate-800 text-slate-200 font-medium text-sm sm:text-base border border-slate-700/60 transition flex items-start gap-3';
        btn.innerHTML = `<span class="w-6 h-6 rounded-full bg-slate-800 border border-slate-700 text-cyan-400 text-xs font-bold flex items-center justify-center shrink-0 mt-0.5">${String.fromCharCode(65 + idx)}</span> <span>${opt}</span>`;
        btn.onclick = () => selectQuizOption(idx);
        optionsContainer.appendChild(btn);
      });

      document.getElementById('quiz-explanation').classList.add('hidden');
      document.getElementById('quiz-next-btn').classList.add('hidden');

      // Timer Setup
      state.quizTimeLeft = 20;
      document.getElementById('quiz-timer').innerText = state.quizTimeLeft;
      clearInterval(state.quizTimerInterval);
      state.quizTimerInterval = setInterval(() => {
        state.quizTimeLeft--;
        document.getElementById('quiz-timer').innerText = state.quizTimeLeft;
        if (state.quizTimeLeft <= 0) {
          clearInterval(state.quizTimerInterval);
          selectQuizOption(-1); // Timeout
        }
      }, 1000);
    }

    function selectQuizOption(idx) {
      clearInterval(state.quizTimerInterval);
      const q = quizData[state.currentQuizIndex];
      const buttons = document.getElementById('quiz-options').children;

      for (let i = 0; i < buttons.length; i++) {
        buttons[i].disabled = true;
        if (i === q.correct) {
          buttons[i].classList.add('bg-emerald-500/20', 'border-emerald-500', 'text-emerald-300');
        } else if (i === idx) {
          buttons[i].classList.add('bg-rose-500/20', 'border-rose-500', 'text-rose-300');
        }
      }

      if (idx === q.correct) {
        playSound('correct');
        const points = 100 + state.quizTimeLeft * 5;
        updateScore(points);
      } else {
        playSound('wrong');
      }

      document.getElementById('quiz-explanation-text').innerText = q.explanation;
      document.getElementById('quiz-explanation').classList.remove('hidden');
      document.getElementById('quiz-next-btn').classList.remove('hidden');
    }

    function nextQuizQuestion() {
      state.currentQuizIndex++;
      if (state.currentQuizIndex < quizData.length) {
        loadQuizQuestion();
      } else {
        switchMode('finish');
      }
    }

      const choicesContainer = document.getElementById('sim-method-choices');
      choicesContainer.innerHTML = '';

      c.choices.forEach((choice, idx) => {

        btn.className = 'w-full text-left p-4 rounded-xl glass-card hover:border-purple-500/50 text-slate-200 font-semibold text-sm transition flex items-center justify-between group';

      document.getElementById('sim-feedback').classList.add('hidden');
      document.getElementById('sim-next-btn').classList.add('hidden');

    function selectSimChoice(choice) {
      const feedback = document.getElementById('sim-feedback');
      const title = document.getElementById('sim-feedback-title');
      const body = document.getElementById('sim-feedback-body');

      feedback.classList.remove('hidden', 'bg-emerald-950/40', 'border-emerald-500/40', 'bg-rose-950/40', 'border-rose-500/40');

      if (choice.correct) {
        playSound('correct');
        updateScore(150);
        feedback.classList.add('bg-emerald-950/40', 'border-emerald-500/40');
        title.className = 'font-bold flex items-center gap-2 text-emerald-400';
        title.innerHTML = '<i class="fa-solid fa-circle-check"></i> Точный диагноз поставлен!';
      } else {
        playSound('wrong');
        feedback.classList.add('bg-rose-950/40', 'border-rose-500/40');
        title.className = 'font-bold flex items-center gap-2 text-rose-400';
        title.innerHTML = '<i class="fa-solid fa-triangle-exclamation"></i> Неоптимальный выбор';
      }

      body.innerText = choice.reason;
      document.getElementById('sim-next-btn').classList.remove('hidden');

      // Disable choice buttons
      const buttons = document.getElementById('sim-method-choices').children;
      for (let b of buttons) b.disabled = true;
    }

    function startMatchingChallenge() {
      switchMode('match');
      state.selectedMatchItem = null;
      state.matchAnswers = {};

      // Render categories
      const targetContainer = document.getElementById('match-targets');
      targetContainer.innerHTML = '';

      matchCategories.forEach(cat => {
        const box = document.createElement('div');
        box.className = `glass-card border-t-4 ${cat.border} rounded-2xl p-4 min-h-[160px] flex flex-col justify-start cursor-pointer hover:bg-slate-800/80 transition`;
        box.onclick = () => assignToCategory(cat.id);
        box.innerHTML = `
          <h4 class="font-bold text-white text-base border-b border-slate-700/60 pb-2 mb-3 text-center">${cat.title}</h4>
          <div id="cat-box-${cat.id}" class="space-y-2 flex-grow"></div>
        `;
        targetContainer.appendChild(box);
      });

      renderItemPool();
    }

    function renderItemPool() {
      const pool = document.getElementById('match-item-pool');
      pool.innerHTML = '';

      matchItems.forEach(item => {
        if (!state.matchAnswers[item.id]) {
          const chip = document.createElement('div');
          chip.className = `px-3 py-2 rounded-xl text-xs sm:text-sm font-semibold border transition cursor-pointer ${
            state.selectedMatchItem === item.id 
              ? 'bg-pink-500 text-white border-pink-400 shadow-lg shadow-pink-500/30 scale-105' 
              : 'bg-slate-800 text-slate-200 border-slate-700 hover:border-pink-500/50'
          }`;
          chip.innerText = item.text;
          chip.onclick = (e) => {
            e.stopPropagation();
            playSound('click');
            state.selectedMatchItem = item.id;
            renderItemPool();
          };
          pool.appendChild(chip);
        }
      });
    }

    function assignToCategory(catId) {
      if (!state.selectedMatchItem) return;
      playSound('click');
      state.matchAnswers[state.selectedMatchItem] = catId;
      const itemId = state.selectedMatchItem;
      state.selectedMatchItem = null;

      // Update UI
      const itemObj = matchItems.find(i => i.id === itemId);
      const catBox = document.getElementById(`cat-box-${catId}`);
      
      const card = document.createElement('div');
      card.className = 'p-2 rounded-lg bg-slate-900 border border-slate-700 text-xs text-slate-200 flex justify-between items-center';
      card.innerHTML = `<span>${itemObj.text}</span> <i class="fa-solid fa-xmark text-slate-500 hover:text-red-400 cursor-pointer ml-1" onclick="unassignItem('${itemId}', event)"></i>`;
      catBox.appendChild(card);

      renderItemPool();
    }

    function unassignItem(itemId, e) {
      e.stopPropagation();
      playSound('click');
      delete state.matchAnswers[itemId];
      startMatchingChallenge(); // Re-render whole view
    }

    function checkMatchResults() {
      let correctCount = 0;
      matchItems.forEach(item => {
        if (state.matchAnswers[item.id] === item.category) {
          correctCount++;
        }
      });

      if (correctCount === matchItems.length) {
        playSound('correct');
        updateScore(300);
        alert(`Великолепно! Все ${correctCount} признака верно распределены по методам! (+300 баллов)`);
        switchMode('finish');
      } else {
        playSound('wrong');
        alert(`Правильно совпало: ${correctCount} из ${matchItems.length}. Попробуйте скорректировать распределение!`);
      }
    }
